## 1. 概述

### 1.1 问题陈述

DevAgent V1 的工具层仅包含 3 个有效工具（FileTool、PytestRunner、PatchTool），严重限制了 Agent 的操作能力。业界顶级 SWE Agent 均配备了丰富的终端级工具集。

### 1.2 设计目标

- 从 3 个工具扩展到 10+ 个工具
- 每个工具都提供 JSON Schema 定义，支持 LLM function calling
- 工具具备安全边界（沙箱化、权限控制）
- 工具输出结构化且可被 LLM 理解

## 2. 工具分类与设计

### 2.1 工具总览

```
工具分类体系：
```

### 2.2 各类工具详细设计

#### 2.2.1 file_read（文件读取）

```python
TOOL_SCHEMA = {
    "name": "file_read",
    "description": "Read a file from the workspace. Shows line numbers.",
    "parameters": {
        "type": "object",
        "properties": {
            "path": {
                "type": "string",
                "description": "Relative or absolute path to the file"
            },
            "offset": {
                "type": "integer",
                "description": "Line number to start reading from (1-indexed)"
            },
            "limit": {
                "type": "integer",
                "description": "Maximum number of lines to read"
            }
        },
        "required": ["path"]
    }
}

class FileReadTool(Tool):
    async def execute(self, params, workspace):
        full_path = self._resolve_path(params["path"], workspace)
        content = Path(full_path).read_text(encoding="utf-8")
        lines = content.split("\n")
        total = len(lines)

        if params.get("offset"):
            offset = params["offset"] - 1
            lines = lines[offset:]
        if params.get("limit"):
            lines = lines[:params["limit"]]

        # 返回带行号的内容
        output = "\n".join(f"{i+1:4d}|{line}" for i, line in enumerate(lines))
        return ToolResult(
            success=True,
            output=output,
            structured={"path": str(full_path), "total_lines": total, "shown_lines": len(lines)}
        )
```

#### 2.2.2 file_edit（精确编辑）

```python
TOOL_SCHEMA = {
    "name": "file_edit",
    "description": """Edit a file by replacing exact text.
The old_string must match EXACTLY once in the file (including whitespace).
If you want to replace multiple occurrences, use replace_all=true.""",
    "parameters": {
        "type": "object",
        "properties": {
            "path": {"type": "string", "description": "File path to edit"},
            "old_string": {"type": "string", "description": "Exact text to replace"},
            "new_string": {"type": "string", "description": "Replacement text"},
            "replace_all": {
                "type": "boolean",
                "default": False,
                "description": "Replace all occurrences"
            }
        },
        "required": ["path", "old_string", "new_string"]
    }
}

class FileEditTool(Tool):
    async def execute(self, params, workspace):
        full_path = self._resolve_path(params["path"], workspace)
        content = Path(full_path).read_text(encoding="utf-8")
        old = params["old_string"]
        new = params["new_string"]

        # 检查匹配次数
        count = content.count(old)
        if count == 0:
            return ToolResult(False, error=f"old_string not found in {params['path']}")
        if count > 1 and not params.get("replace_all"):
            return ToolResult(False, error=f"old_string occurs {count} times. Use replace_all=true or make it more specific")

        if params.get("replace_all"):
            new_content = content.replace(old, new)
        else:
            new_content = content.replace(old, new, 1)

        Path(full_path).write_text(new_content, encoding="utf-8")

        # 生成 diff 展示给 LLM
        diff = self._generate_diff(content, new_content, params["path"])
        return ToolResult(success=True, output=f"Edited {params['path']}\n\n{diff}")
```

#### 2.2.3 grep_text（文本搜索）

```python
TOOL_SCHEMA = {
    "name": "grep_text",
    "description": "Search for text/pattern in workspace files. Use BEFORE reading files to locate relevant code.",
    "parameters": {
        "type": "object",
        "properties": {
            "pattern": {"type": "string", "description": "Text or regex pattern"},
            "path": {"type": "string", "description": "Directory to search (default: workspace root)"},
            "include": {"type": "string", "description": "File pattern filter, e.g. '*.py'"},
            "context_lines": {"type": "integer", "default": 2, "description": "Lines of context around matches"}
        },
        "required": ["pattern"]
    }
}

class GrepTextTool(Tool):
    async def execute(self, params, workspace):
        pattern = params["pattern"]
        search_path = self._resolve_path(params.get("path", workspace), workspace)
        include = params.get("include", "*.py")

        # 使用 ripgrep 或 Python fallback
        matches = []
        for file_path in Path(search_path).rglob(include):
            if self._should_skip(file_path):
                continue
            try:
                content = file_path.read_text(encoding="utf-8")
                for i, line in enumerate(content.split("\n"), 1):
                    if re.search(pattern, line):
                        matches.append({
                            "file": str(file_path.relative_to(workspace)),
                            "line": i,
                            "content": line.strip()
                        })
            except Exception:
                continue

        # 限制结果数量
        if len(matches) > 50:
            matches = matches[:50]
            truncated = True
        else:
            truncated = False

        output = self._format_matches(matches, truncated)
        return ToolResult(success=True, output=output,
                         structured={"count": len(matches), "matches": matches})
```

#### 2.2.4 grep_ast（AST 结构化搜索）

```python
TOOL_SCHEMA = {
    "name": "grep_ast",
    "description": "Search for code patterns using AST queries. E.g., find all function definitions, class hierarchies, call sites.",
    "parameters": {
        "type": "object",
        "properties": {
            "pattern": {
                "type": "string",
                "description": "AST pattern type: function_def | class_def | function_call(name) | import | assignment"
            },
            "name": {"type": "string", "description": "Optional name filter (e.g. specific function name)"},
            "path": {"type": "string", "description": "Directory to search"}
        },
        "required": ["pattern"]
    }
}

class GrepASTTool(Tool):
    PREDEFINED_QUERIES = {
        "function_def": lambda tree: [n.name for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)],
        "class_def": lambda tree: [n.name for n in ast.walk(tree) if isinstance(n, ast.ClassDef)],
        "function_call": lambda tree, name: [
            (n.func.id if isinstance(n.func, ast.Name) else n.func.attr)
            for n in ast.walk(tree)
            if isinstance(n, ast.Call) and (
                not name or (isinstance(n.func, ast.Name) and n.func.id == name))
        ],
        "import": lambda tree: [
            (n.module, [alias.name for alias in n.names])
            for n in ast.walk(tree) if isinstance(n, ast.Import)
        ] + [
            (n.module, [n.name for n in n.names])
            for n in ast.walk(tree) if isinstance(n, ast.ImportFrom)
        ],
    }
```

#### 2.2.5 shell_run（Shell 命令执行）

```python
TOOL_SCHEMA = {
    "name": "shell_run",
    "description": """Run a shell command in the workspace.
Use for: running tests, installing dependencies, checking syntax,
building the project, executing the application.

Never use for destructive operations like rm -rf or force push.""",
    "parameters": {
        "type": "object",
        "properties": {
            "command": {"type": "string", "description": "Shell command to execute"},
            "timeout": {"type": "integer", "default": 120, "description": "Timeout in seconds"}
        },
        "required": ["command"]
    }
}

class ShellRunTool(Tool):
    DANGEROUS_PATTERNS = [
        r"rm\s+-rf\s+/", r"git\s+push\s+--force",
        r"sudo\s+", r">\s*/dev/sda", r"mkfs\."
    ]

    async def execute(self, params, workspace):
        command = params["command"]
        timeout = params.get("timeout", 120)

        # 安全检查
        for pattern in self.DANGEROUS_PATTERNS:
            if re.search(pattern, command):
                return ToolResult(False, error=f"Blocked dangerous command pattern: {pattern}")

        try:
            proc = await asyncio.create_subprocess_shell(
                command,
                cwd=workspace,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            stdout, stderr = await asyncio.wait_for(
                proc.communicate(), timeout=timeout
            )
            return ToolResult(
                success=proc.returncode == 0,
                output=stdout.decode()[:5000] or "(no output)",
                structured={
                    "returncode": proc.returncode,
                    "stderr": stderr.decode()[:2000]
                }
            )
        except asyncio.TimeoutError:
            return ToolResult(False, error=f"Command timed out after {timeout}s")
```

## 3. 工具注册与发现

```python
class ToolRegistry:
    """工具注册中心，管理所有可用工具"""

    def __init__(self):
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool):
        self._tools[tool.name] = tool

    def get_openai_schemas(self) -> list[dict]:
        return [t.to_openai_schema() for t in self._tools.values()]

    def get_descriptions(self) -> str:
        """生成给 LLM 的工具说明文本"""
        lines = []
        for t in self._tools.values():
            lines.append(f"## {t.name}")
            lines.append(f"{t.description}")
            lines.append(f"Parameters: {json.dumps(t.parameters, indent=2)}")
            lines.append("")
        return "\n".join(lines)

    async def execute(self, name: str, params: dict, workspace: str) -> ToolResult:
        tool = self._tools.get(name)
        if not tool:
            return ToolResult(False, error=f"Unknown tool: {name}. Available: {list(self._tools)}")
        try:
            return await tool.execute(params, workspace)
        except Exception as e:
            return ToolResult(False, error=f"Tool {name} failed: {str(e)}")
```

## 4. 安全设计

### 4.1 分层安全模型

```
L1: 命令白名单 —— shell_run 拒绝危险模式
L2: 路径沙箱化 —— 所有文件操作限制在 workspace 内
L3: 资源限制   —— 超时、内存限制、文件大小限制
L4: 审计日志   —— 所有工具调用记录到日志
```

### 4.2 路径沙箱

```python
class PathSandbox:
    def __init__(self, workspace: str):
        self.workspace = Path(workspace).resolve()

    def resolve(self, path: str) -> Path:
        full = (self.workspace / path).resolve()
        if not str(full).startswith(str(self.workspace)):
            raise PermissionError(f"Path escapes workspace: {path}")
        return full
```

## 5. 工具输出格式规范

所有工具输出遵循统一格式，便于 LLM 理解：

```
[TOOL: {tool_name}]
STATUS: {SUCCESS|ERROR}
OUTPUT:
{formatted_output}
[END: {tool_name}]
```
