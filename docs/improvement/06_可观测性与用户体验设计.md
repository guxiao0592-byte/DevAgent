## 1. 概述

### 1.1 问题陈述

DevAgent V1 的可观测性和用户体验存在以下问题：
- 执行过程不可见：用户只能看到开始和结束，中间过程是黑盒
- WebSocket 流是模拟的（`phases = ["initializing", "analyzing", ...]`），非真实事件
- 无历史任务管理和对比
- 无法在 Agent 执行过程中介入和引导
- 错误信息不结构化，调试困难

### 1.2 设计目标

- 实现真实的流式事件推送（每个工具调用、每次状态变化）
- 提供交互式 Web Dashboard
- 支持人类在环（Human-in-the-Loop）的确认和引导
- 结构化日志和可回放执行轨迹
- 任务历史管理与趋势分析

### 2.2 事件结构

```python
@dataclass
class DevAgentEvent:
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    type: EventType
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    task_id: str
    iteration: int  # 当前迭代序号

    # 事件数据（根据类型不同）
    data: dict = field(default_factory=dict)

    # 关联信息
    parent_event_id: str | None = None
    tool_call_id: str | None = None

    def to_sse(self) -> str:
        """转换为 Server-Sent Events 格式"""
        return f"event: {self.type.value}\ndata: {json.dumps(self.to_dict())}\n\n"

    def to_dict(self) -> dict:
        return {
            "id": self.id,
            "type": self.type.value,
            "timestamp": self.timestamp,
            "task_id": self.task_id,
            "iteration": self.iteration,
            "data": self.data
        }
```

### 2.3 事件总线

```python
class EventBus:
    """发布-订阅事件总线，支持同步和异步订阅者"""

    def __init__(self):
        self._sync_subscribers: dict[EventType, list[Callable]] = defaultdict(list)
        self._async_subscribers: dict[EventType, list[Callable]] = defaultdict(list)
        self._all_events: list[DevAgentEvent] = []  # 完整事件日志

    def subscribe(self, event_type: EventType, handler: Callable, async_mode=False):
        if async_mode:
            self._async_subscribers[event_type].append(handler)
        else:
            self._sync_subscribers[event_type].append(handler)

    async def publish(self, event: DevAgentEvent):
        self._all_events.append(event)

        # 同步处理
        for handler in self._sync_subscribers.get(event.type, []):
            handler(event)

        # 异步处理
        for handler in self._async_subscribers.get(event.type, []):
            asyncio.create_task(handler(event))

    def subscribe_all(self, handler: Callable):
        """订阅所有事件类型"""
        for event_type in EventType:
            self.subscribe(event_type, handler)
```

## 4. 人类在环（Human-in-the-Loop）

### 4.1 介入点设计

```python
class HumanInTheLoop:
    """定义 Agent 需要人类介入的场景"""

    INTERVENTION_POINTS = {
        "destructive_operation": {
            "description": "即将执行可能破坏性的操作",
            "triggers": [
                "file_write with delete mode",
                "shell_run with rm/git reset --hard",
            ],
            "requires_approval": True
        },
        "large_edit": {
            "description": "即将修改超过 50 行代码",
            "triggers": ["file_edit with more than 50 lines changed"],
            "requires_approval": True
        },
        "uncertain_fix": {
            "description": "Agent 对修复方案不确定（confidence < 0.5）",
            "triggers": ["repair with confidence < 0.5"],
            "requires_approval": True
        },
        "strategy_choice": {
            "description": "存在多个可行方案，需要选择",
            "triggers": ["multiple fix hypotheses with similar scores"],
            "requires_approval": True
        },
        "max_iterations_near": {
            "description": "接近最大迭代次数，可能无法完成任务",
            "triggers": ["iteration >= max_iterations * 0.8"],
            "requires_approval": False  # 仅提示
        }
    }

    async def check_intervention(self, event: DevAgentEvent) -> InterventionResult:
        """检查是否需要人类介入"""
        for point_id, config in self.INTERVENTION_POINTS.items():
            if self._matches_trigger(event, config["triggers"]):
                if config["requires_approval"]:
                    # 暂停 Agent，等待用户批准
                    approval = await self._request_approval(point_id, event)
                    return InterventionResult(
                        intervention_needed=True,
                        point=point_id,
                        approved=approval
                    )
                else:
                    # 仅通知用户，不暂停
                    await self._notify_user(point_id, event)
        return InterventionResult(intervention_needed=False)

    async def _request_approval(self, point_id: str, event: DevAgentEvent) -> bool:
        """通过 WebSocket 向 Dashboard 发送审批请求"""
        approval_event = DevAgentEvent(
            type=EventType.HUMAN_APPROVAL_REQUESTED,
            data={
                "point": point_id,
                "details": event.data,
                "timeout_seconds": 300  # 5 分钟超时后自动拒绝
            }
        )
        await self.event_bus.publish(approval_event)

        # 等待用户响应
        try:
            response = await asyncio.wait_for(
                self._wait_for_approval(approval_event.id),
                timeout=300
            )
            return response
        except asyncio.TimeoutError:
            return False  # 超时自动拒绝
```

### 4.2 用户引导指令

```python
class UserGuidance:
    """用户可以在任务执行中发送引导指令"""

    GUIDANCE_COMMANDS = {
        "focus_on": "引导 Agent 关注特定文件/函数",
        "skip_step": "跳过当前步骤",
        "try_alternative": "建议 Agent 尝试备选方案",
        "increase_budget": "增加最大迭代次数",
        "rollback": "回退到上一个检查点",
    }

    async def handle_guidance(self, command: dict, state: AgentLoopState):
        """处理用户的引导指令"""
        cmd_type = command.get("type")
        if cmd_type == "focus_on":
            # 将用户指定的文件加入焦点上下文
            state.add_priority_file(command["file"])
        elif cmd_type == "try_alternative":
            # 标记当前方案为已尝试，促使尝试备选
            state.mark_attempted(state.current_approach)
        elif cmd_type == "rollback":
            # 从检查点恢复
            await state.checkpoint_manager.restore(command.get("checkpoint_id"))
```

## 6. 执行回放

```python
class ExecutionReplay:
    """基于事件日志的执行回放"""

    async def replay(self, task_id: str, speed: float = 1.0):
        """回放指定任务的执行过程"""
        events_path = f".devagent/tasks/{task_id}/events.jsonl"

        with open(events_path) as f:
            events = [json.loads(line) for line in f]

        # 计算真实的时间间隔
        for i, event in enumerate(events):
            if i > 0:
                prev_time = datetime.fromisoformat(events[i-1]["timestamp"])
                curr_time = datetime.fromisoformat(event["timestamp"])
                delay = (curr_time - prev_time).total_seconds() / speed
                await asyncio.sleep(delay)

            # 发送事件到 Dashboard（像真实执行一样）
            yield DevAgentEvent.from_dict(event)
```
