## 1. 问题陈述

DevAgent 当前是纯文本 Agent，无法处理视觉信息：
- 用户发来错误截图，Agent 无法解读
- UI bug（布局错乱、颜色问题）对纯文本 Agent 不透明
- 无法从架构图、流程图等视觉文档中提取信息

## 2. 设计方案

### 2.2 ImageTool 设计

```python
class ImageReadTool(BaseTool):
    """读取并分析图片内容"""
    name = "image_read"
    description = "Read and analyze an image file. Supports screenshots, diagrams, and UI mockups."
    parameters = {
        "path": {
            "type": "string",
            "description": "Path to image file (PNG, JPG, GIF, WebP)"
        },
        "query": {
            "type": "string",
            "description": "What to look for in the image (e.g., 'error message', 'layout issue')"
        }
    }

    SUPPORTED_FORMATS = {".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp"}

    async def execute(self, params: dict, workspace: str) -> ToolResult:
        sandbox = PathSandbox(workspace)
        file_path = sandbox.resolve(params["path"])
        query = params.get("query", "Describe this image in detail")

        if not file_path.exists():
            return ToolResult(False, error=f"Image not found: {params['path']}")

        ext = file_path.suffix.lower()
        if ext not in self.SUPPORTED_FORMATS:
            return ToolResult(False, error=f"Unsupported format: {ext}. Supported: {self.SUPPORTED_FORMATS}")

        # Encode image as base64 for LLM
        import base64
        image_data = base64.b64encode(file_path.read_bytes()).decode()

        # Call multimodal LLM
        result = await self._analyze_image(image_data, ext, query)
        return result

    async def _analyze_image(self, image_b64: str, ext: str, query: str) -> ToolResult:
        """使用多模态 LLM 分析图片"""
        try:
            # For OpenAI GPT-4V compatible API
            response = self.llm.chat_with_image(
                messages=[{
                    "role": "user",
                    "content": [
                        {"type": "text", "text": query},
                        {"type": "image_url",
                         "image_url": {"url": f"data:image/{ext[1:]};base64,{image_b64}"}}
                    ]
                }],
                max_tokens=1000
            )
            return ToolResult(True, response,
                             structured={"format": ext, "query": query})
        except Exception as e:
            return ToolResult(False, error=f"Image analysis failed: {e}")
```

### 2.3 多模态上下文构建

```python
class MultimodalContextBuilder:
    """在上下文中插入图片描述，让纯文本 Agent 也能利用视觉信息"""

    def build(self, images: list[ImageDescription],
              task: str) -> list[dict]:
        """将图片分析结果注入上下文"""
        if not images:
            return []

        parts = ["## Image Analysis Results"]
        for i, img in enumerate(images, 1):
            parts.append(f"""### Image {i}: {img.path}
- Type: {img.detected_type}  (error_screenshot | ui_mockup | architecture_diagram | log_output)
- Summary: {img.summary}
- Key details: {img.key_details}
- Relevant code: {img.relevant_code_hint or 'N/A'}
""")

        return [{"role": "user", "content": "\n".join(parts)}]
```

### 2.4 Screenshot 专用分析器

```python
class ScreenshotAnalyzer:
    """针对软件工程截图的专门分析逻辑"""

    ANALYSIS_PROMPTS = {
        "error_screenshot": """This is a screenshot of an error. Analyze:
1. What is the exact error message?
2. What file/line is mentioned?
3. Is this a runtime error, build error, or test failure?
4. What is the likely root cause?""",

        "ui_mockup": """This is a UI mockup or screenshot. Analyze:
1. What UI components are shown?
2. Are there any layout issues (overlap, misalignment)?
3. Does it match the described requirements?
4. What CSS/component changes would fix visual issues?""",

        "architecture_diagram": """This is an architecture or design diagram. Analyze:
1. What components/modules are shown?
2. What are the data flows?
3. Does the current codebase match this design?
4. What is missing or inconsistent?""",

        "log_output": """This is a log or terminal output screenshot. Analyze:
1. What errors or warnings are shown?
2. What is the sequence of events?
3. What is the root cause indicated?"""
    }

    async def analyze(self, image_b64: str, detected_type: str) -> ImageDescription:
        prompt = self.ANALYSIS_PROMPTS.get(detected_type,
                                           "Describe this image in detail")
        response = await self._call_multimodal_llm(image_b64, prompt)
        return self._parse_response(response)
```

### 2.5 模型选择策略

```python
class MultimodalModelRouter:
    """根据可用性和任务复杂度选择多模态模型"""

    MODELS = [
        {"name": "gpt-4o", "provider": "openai", "quality": "best", "cost": "high"},
        {"name": "claude-sonnet-4-6", "provider": "anthropic", "quality": "best", "cost": "high"},
        {"name": "gpt-4o-mini", "provider": "openai", "quality": "good", "cost": "low"},
        {"name": "deepseek-vl2", "provider": "deepseek", "quality": "adequate", "cost": "low"},
    ]

    @classmethod
    def select(cls, task_complexity: str, available_providers: list[str]) -> str:
        """选择最合适的多模态模型"""
        candidates = [m for m in cls.MODELS
                     if m["provider"] in available_providers]
        if task_complexity == "high":
            candidates = [m for m in candidates if m["quality"] == "best"]
        elif task_complexity == "low":
            candidates = [m for m in candidates if m["cost"] == "low"]
        return candidates[0]["name"] if candidates else None

    @classmethod
    def is_multimodal_available(cls, llm_config: dict) -> bool:
        """检查当前 LLM 配置是否支持多模态"""
        provider = llm_config.get("provider", "")
        model = llm_config.get("model", "")
        multimodal_models = {"gpt-4o", "gpt-4o-mini", "gpt-4-turbo",
                            "claude-3-opus", "claude-3-sonnet",
                            "claude-sonnet-4-6", "claude-opus-4-7",
                            "deepseek-vl2"}
        return model in multimodal_models
```
