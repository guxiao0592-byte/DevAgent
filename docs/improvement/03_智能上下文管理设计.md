## 1. 概述

### 1.1 问题陈述

DevAgent V1 的上下文管理仅使用 `_truncate_text(content, max_chars=4000)` 进行简单截断，存在以下严重问题：
- 大项目中 4000 字符只能覆盖 1-2 个文件
- 与 bug 无关的代码占据珍贵的上下文窗口
- 跨文件依赖关系丢失，Agent 无法理解调用链
- 无历史压缩，多轮对话中早期信息被丢弃

### 1.2 设计目标

- 将 LLM 上下文窗口有效利用率从 ~30% 提升到 >80%
- 支持 10 万+ 行代码的仓库
- 保留跨文件依赖和调用链信息
- 历史压缩不丢失关键决策信息

## 2. 核心设计：分层上下文模型

### 2.1 模型架构

```
L1: Repo Map（常驻上下文，~500 tokens）
    └── 仓库的结构化地图：目录树 + 模块依赖图 + 关键符号表

L2: Relevant Files（按需加载，~3000 tokens）
    └── 与当前任务最相关的 3-5 个文件的签名摘要

L3: Focused Context（当前焦点，~4000 tokens）
    └── 正在操作的文件的完整内容 + 直接依赖

L4: History Summary（压缩历史，动态）
    └── 早期轮次的压缩摘要 + 最近 K 轮完整历史
```

### 2.2 各层详细设计

#### L1: Repo Map（仓库地图）

```python
class RepoMap:
    """仓库的结构化地图，作为 Agent 的全局导航"""

    def generate(self, repo_path: str) -> str:
        """生成仓库地图"""
        sections = []

        # 1. 目录结构（仅关键目录）
        tree = self._generate_tree(repo_path, max_depth=3)

        # 2. 模块依赖图
        deps = self._extract_imports(repo_path)

        # 3. 关键符号表（类、函数、常量）
        symbols = self._extract_top_level_symbols(repo_path)

        # 4. 测试与源码映射
        test_map = self._map_tests_to_source(repo_path)

        return self._format(template="repo_map", **locals())

    def _generate_tree(self, path, max_depth):
        """生成简化的目录树"""
        # 忽略: __pycache__, .git, venv, node_modules, egg-info
        # 展开: src/, tests/, configs/
        pass

    def _extract_imports(self, path):
        """提取模块间依赖关系"""
        # 解析所有 .py 文件的 import 语句
        # 构建：{module: [depends_on]}
        deps = {}
        for py_file in Path(path).rglob("*.py"):
            tree = ast.parse(py_file.read_text())
            imports = self._find_imports(tree)
            module = str(py_file.relative_to(path))
            for imp in imports:
                if imp in deps:
                    deps[imp].append(module)
                else:
                    deps[imp] = [module]
        return deps

    def _extract_top_level_symbols(self, path):
        """提取所有顶层符号（类、函数）"""
        symbols = []
        for py_file in Path(path).rglob("*.py"):
            tree = ast.parse(py_file.read_text())
            for node in ast.iter_child_nodes(tree):
                if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
                    symbols.append({
                        "name": node.name,
                        "type": "class" if isinstance(node, ast.ClassDef) else "function",
                        "file": str(py_file.relative_to(path)),
                        "lineno": node.lineno
                    })
        return symbols
```

#### L2: 智能文件检索

```python
class RelevantFileRetriever:
    """基于多种信号检索与任务相关的文件"""

    def retrieve(self, task: str, repo_path: str, top_k: int = 5) -> list[str]:
        scores = {}

        # Signal 1: 关键词匹配（快速但是粗糙）
        keywords = self._extract_keywords(task)
        for file_path in self._list_source_files(repo_path):
            scores[file_path] = self._keyword_score(file_path, keywords)

        # Signal 2: 语义相似度（基于嵌入向量，精确但是慢）
        task_embedding = self._embed(task)
        for file_path, file_embedding in self._file_embeddings(repo_path).items():
            scores[file_path] += 0.5 * self._cosine_sim(task_embedding, file_embedding)

        # Signal 3: 修改历史（最近修改的文件更可能相关）
        for file_path, last_modified in self._git_recent_files(repo_path).items():
            recency_score = 1.0 / (1.0 + self._days_since(last_modified))
            scores[file_path] += 0.3 * recency_score

        # Signal 4: 测试失败指向（如果已经有失败的测试）
        for file_path, test_failure_score in self._test_failure_signals(repo_path).items():
            scores[file_path] += 0.8 * test_failure_score

        ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [f for f, s in ranked[:top_k]]
```

#### L3: 焦点上下文展开

```python
class FocusContextBuilder:
    """构建当前焦点文件的完整上下文"""

    def build(self, focus_file: str, repo_path: str) -> str:
        sections = []

        # 1. 焦点文件完整内容
        content = Path(repo_path, focus_file).read_text()
        sections.append(f"### {focus_file} (FULL)\n```python\n{content}\n```")

        # 2. 直接依赖的签名（焦点文件 import 了什么）
        imports = self._extract_imports_of_file(focus_file)
        for imported in imports:
            signatures = self._extract_signatures(imported, repo_path)
            if signatures:
                sections.append(f"### {imported} (signatures)\n{signatures}")

        # 3. 直接被依赖者的签名（谁 import 了焦点文件）
        dependents = self._find_dependents(focus_file, repo_path)
        for dep in dependents[:3]:  # 最多 3 个调用者
            signatures = self._extract_signatures(dep, repo_path)
            if signatures:
                sections.append(f"### {dep} (caller)\n{signatures}")

        return "\n\n".join(sections)
```

#### L4: 历史压缩

```python
class HistoryCompressor:
    """压缩早期对话历史以节省上下文空间"""

    COMPRESSION_PROMPT = """Summarize the following agent action history.
Preserve: file modifications, test results, error messages, key decisions.
Drop: exploratory commands, unchanged read operations.

History:
{history}

Summary:"""

    def compress(self, observations: list[Observation]) -> str:
        """压缩历史记录为精简摘要"""
        history_text = self._format_observations(observations)

        if self._token_count(history_text) < 2000:
            return history_text  # 不需要压缩

        # 用 LLM 压缩历史
        summary = self.llm.chat(
            messages=[{"role": "user", "content": history_text}],
            system_prompt=self.COMPRESSION_PROMPT
        )
        return summary

    def _format_observations(self, obs: list[Observation]) -> str:
        """格式化观察序列为文本"""
        lines = []
        for o in obs:
            lines.append(f"[Step] {o.action.tool_name}({json.dumps(o.action.tool_params)})")
            if o.result.success:
                # 保留输出摘要，丢弃完整输出
                output_summary = self._summarize_output(o.result.output)
                lines.append(f"  Result: OK — {output_summary}")
            else:
                lines.append(f"  Result: ERROR — {o.result.error[:200]}")
        return "\n".join(lines)
```

## 3. 动态上下文预算分配

```python
class ContextBudget:
    """管理 LLM 上下文窗口的预算分配"""

    def __init__(self, max_tokens: int = 8000):
        self.max_tokens = max_tokens
        self.budget = {
            "system_prompt": 500,       # 固定
            "task_description": 300,    # 固定
            "repo_map": 500,            # 固定
            "relevant_files": 2000,     # 按需
            "focus_context": 2000,      # 按需
            "recent_history": 2000,     # 最多
            "compressed_history": 500,  # 剩余空间
        }

    def allocate(self, phase: str) -> dict:
        """根据当前阶段动态调整预算"""
        if phase == "exploration":
            # 探索阶段：更多给文件检索，少给编辑历史
            return {**self.budget, "relevant_files": 3000, "recent_history": 1000}
        elif phase == "editing":
            # 编辑阶段：更多给焦点上下文，少给文件检索
            return {**self.budget, "focus_context": 3000, "relevant_files": 1000}
        elif phase == "verification":
            # 验证阶段：平衡分配，保留测试结果
            return {**self.budget, "recent_history": 2500, "focus_context": 1500}
        return self.budget
```

## 4. 缓存策略

```python
class ContextCache:
    """缓存已读取的文件内容、嵌入向量、签名摘要"""

    def __init__(self):
        self._file_contents: dict[str, tuple[float, str]] = {}  # path → (mtime, content)
        self._embeddings: dict[str, list[float]] = {}
        self._signatures: dict[str, str] = {}
        self._repo_map: str | None = None

    def get_file(self, path: str) -> str:
        """读取文件，缓存带 mtime 验证"""
        mtime = os.path.getmtime(path)
        if path in self._file_contents and self._file_contents[path][0] == mtime:
            return self._file_contents[path][1]
        content = Path(path).read_text()
        self._file_contents[path] = (mtime, content)
        return content

    def get_repo_map(self, path: str) -> str:
        """仓库地图仅首次生成"""
        if self._repo_map is None:
            self._repo_map = RepoMap().generate(path)
        return self._repo_map
```
