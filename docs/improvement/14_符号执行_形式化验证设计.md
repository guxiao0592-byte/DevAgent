## 1. 问题陈述

当 DevAgent 修改关键代码（如认证逻辑、支付计算、数据迁移）时，仅靠单元测试无法保证正确性：
- 测试只能覆盖有限输入
- 无法保证边界条件的完整覆盖
- 安全关键代码的 bug 代价极高

## 2. 设计方案

### 2.1 分层验证策略

```
根据代码重要性选择验证深度：

┌──────────────────┬──────────────────────┐
│ 代码重要性        │ 验证深度              │
├──────────────────┼──────────────────────┤
│ 低（日志、注释）  │ L1: 语法 + Lint      │
│ 中（业务逻辑）    │ L2: 单元测试 + 变异   │
│ 高（认证、支付）  │ L3: 符号执行 + 合约  │
│ 极高（安全关键）  │ L4: 形式化验证       │
└──────────────────┴──────────────────────┘
```

### 2.2 代码重要性自动评估

```python
class CodeImportanceClassifier:
    """自动评估代码的关键程度"""

    HIGH_IMPORTANCE_PATTERNS = {
        "auth": ["authenticate", "login", "password", "token", "session",
                 "authorize", "permission", "role", "oauth"],
        "payment": ["payment", "charge", "billing", "invoice", "price",
                    "currency", "transaction", "refund", "wallet"],
        "data_integrity": ["migrate", "migration", "schema", "backup",
                          "restore", "validate", "sanitize", "escape"],
        "security": ["encrypt", "decrypt", "hash", "salt", "crypto",
                    "signature", "verify", "csrf", "xss", "sql_injection"],
    }

    @classmethod
    def classify(cls, file_path: str, function_name: str,
                 code_content: str) -> str:
        """返回: low | medium | high | critical"""
        content_lower = code_content.lower()
        name_lower = function_name.lower()
        path_lower = file_path.lower()

        score = 0

        for category, keywords in cls.HIGH_IMPORTANCE_PATTERNS.items():
            for kw in keywords:
                if kw in name_lower:
                    score += 3
                if kw in content_lower:
                    score += 1
                if kw in path_lower:
                    score += 2

        if score >= 8:
            return "critical"
        elif score >= 5:
            return "high"
        elif score >= 2:
            return "medium"
        return "low"
```

### 2.3 符号执行集成（基于 CrossHair）

```python
class SymbolicExecutor:
    """对关键函数执行符号执行，验证对所有可能输入都满足合约"""

    def __init__(self):
        self._crosshair_available = self._check_crosshair()

    def verify_function(self, file_path: str, function_name: str,
                        contracts: list[Contract]) -> SymbolicReport:
        """
        使用符号执行验证函数合约。

        合约示例:
          Contract(pre="b > 0", post="result > 0",
                  description="divide: if b > 0, result must be > 0")
        """
        if not self._crosshair_available:
            return SymbolicReport(success=False,
                                 message="CrossHair not installed. Install: pip install crosshair-tool")

        # 构建验证脚本
        verification_script = self._build_verification_script(
            file_path, function_name, contracts
        )

        with tempfile.NamedTemporaryFile(suffix=".py", mode="w", delete=False) as f:
            f.write(verification_script)
            script_path = f.name

        try:
            proc = subprocess.run(
                ["crosshair", "check", script_path],
                capture_output=True, text=True, timeout=60
            )
            return self._parse_crosshair_output(proc.stdout, proc.stderr)
        finally:
            os.unlink(script_path)

    def _build_verification_script(self, file_path: str,
                                    function_name: str,
                                    contracts: list[Contract]) -> str:
        """生成 CrossHair 验证脚本"""
        contracts_block = "\n".join(
            f'@crosshair.with_contracts(\n'
            f'    pre=lambda: {c.pre},\n'
            f'    post=lambda result: {c.post}\n'
            f')\n'
            for c in contracts
        )

        return f'''
import sys
sys.path.insert(0, "{os.path.dirname(os.path.abspath(file_path))}")

import crosshair

{contracts_block}
def {function_name}_wrapper(*args, **kwargs):
    """Wrapper that re-imports and calls the target function."""
    from {self._module_name(file_path)} import {function_name}
    return {function_name}(*args, **kwargs)
'''
```

### 2.4 合约提取

```python
class ContractExtractor:
    """从代码注释和类型注解中自动提取合约"""

    CONTRACT_PATTERNS = [
        # @requires: b > 0
        (r'@requires:\s*(.+)', "requires"),
        # @ensures: result > 0
        (r'@ensures:\s*(.+)', "ensures"),
        # @raises: ValueError when b == 0
        (r'@raises:\s*(\w+)\s+when\s+(.+)', "raises"),
        # Type annotation contracts
        (r'def \w+\(.*?(\w+):\s*(int|float|str|bool)', "type_annotation"),
    ]

    def extract(self, function_source: str) -> list[Contract]:
        contracts = []
        for pattern, contract_type in self.CONTRACT_PATTERNS:
            for m in re.finditer(pattern, function_source, re.MULTILINE):
                if contract_type == "requires":
                    contracts.append(Contract(pre=m.group(1)))
                elif contract_type == "ensures":
                    contracts.append(Contract(post=m.group(1)))
                elif contract_type == "raises":
                    exc_type = m.group(1)
                    condition = m.group(2)
                    contracts.append(Contract(
                        pre=f"not ({condition})" if "not" not in condition else condition,
                        raises=m.group(1)
                    ))
        return contracts
```

### 2.5 LLM 增强的合约生成

```python
class LLMContractGenerator:
    """使用 LLM 为函数自动生成可能的合约，然后人工或自动选择"""

    PROMPT = """You are a formal verification expert. Given the following function, propose pre/post-condition contracts.

## Function
```python
{function_code}
```

## Context
{context}

## Instructions
Propose 2-4 pre/post-condition pairs for this function.
- Pre-condition: what must be true before calling (e.g., "b > 0")
- Post-condition: what must be true after calling (e.g., "result >= 0")
- Each contract must be expressed as a simple Python boolean expression.

Output JSON:
{{
  "contracts": [
    {{
      "pre": "b != 0",
      "post": "isinstance(result, float)",
      "description": "Division by non-zero returns float"
    }}
  ],
  "confidence": "high|medium|low"
}}"""

    def generate(self, function_code: str, context: str = "") -> list[Contract]:
        prompt = self.PROMPT.format(function_code=function_code, context=context)
        result = self.llm.chat_structured(
            messages=[{"role": "user", "content": prompt}]
        )
        contracts = []
        for c in result.get("contracts", []):
            contracts.append(Contract(
                pre=c.get("pre", "True"),
                post=c.get("post", "True"),
                description=c.get("description", "")
            ))
        return contracts
```

### 2.6 验证门集成

```python
class VerificationGate:
    """在 QualityGateSystem 中增加 L5_VERIFY 门"""

    GATE_L5_VERIFY = {
        "name": "Symbolic Verification",
        "blocking": False,  # 非阻断（形式化验证耗时）
        "auto_fix": False,
        "trigger": lambda state: CodeImportanceClassifier.classify(
            state.get("latest_modified_file", ""),
            state.get("latest_modified_function", ""),
            state.get("latest_modified_code", "")
        ) in ("high", "critical"),
    }
```

verification:
  symbolic_execution:
