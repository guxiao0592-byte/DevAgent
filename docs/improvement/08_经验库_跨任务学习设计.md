## 1. 问题陈述

DevAgent 每次执行任务都从零开始，即使遇到过相同或相似的 bug 模式。这导致：
- 重复的探索开销
- 相同错误反复试错
- 无法利用团队历史经验

## 2. 业界参考

| 系统 | 经验复用方式 |
|------|------------|
| SWE-Fixer | 检索相似 bug 的修复 patch 作为 few-shot 示例 |
| AutoCodeRover | 检索相似 issue 的修复历史 |
| Claude Code | Memory 系统：用户/项目/反馈三类持久记忆 |
| CodePlan | 从历史 PR 学习编辑序列模式 |

### 3.2 经验记录结构

```python
@dataclass
class Experience:
    """一条可检索的经验记录"""
    id: str
    type: str  # bug_fix | strategy | refactor | exploration

    # 索引字段（用于相似度匹配）
    bug_signature: str       # 错误摘要："ZeroDivisionError in divide() when b=0"
    error_type: str          # ZeroDivisionError | AttributeError | TypeError | ...
    language: str            # python | javascript | ...
    file_pattern: str        # "**/math_ops.py"
    code_context_before: str # bug 附近的代码片段

    # 结果字段
    fix_description: str     # 修复的简要说明
    fix_patch: str           # unified diff
    success: bool
    iterations_needed: int

    # 元数据
    task_id: str
    timestamp: str
    project: str
    tags: list[str]          # ["division", "boundary", "zero-check"]
```

### 3.3 检索流程

```python
class ExperienceRetriever:
    """检索与当前任务最相关的历史经验"""

    def __init__(self, embedding_model: str = "text-embedding-3-small"):
        self.embedding_model = embedding_model
        self.store: list[Experience] = []
        self.embeddings: dict[str, list[float]] = {}

    def retrieve(self, query: dict, top_k: int = 5) -> list[Experience]:
        """混合检索：语义相似度 + 关键词 + 结构化过滤"""
        scores = {}

        # 1. 语义相似度
        query_text = self._build_query_text(query)
        query_emb = self._embed(query_text)
        for exp_id, emb in self.embeddings.items():
            scores[exp_id] = 0.6 * self._cosine_sim(query_emb, emb)

        # 2. 关键词匹配（加权）
        error_type = query.get("error_type", "")
        keywords = self._extract_keywords(query.get("error_message", ""))
        for exp in self.store:
            if exp.error_type == error_type:
                scores[exp.id] += 0.3
            for kw in keywords:
                if kw in exp.bug_signature.lower():
                    scores[exp.id] += 0.1

        # 3. 结构化过滤（同语言、同项目优先）
        for exp in self.store:
            if exp.language == query.get("language"):
                scores[exp.id] += 0.05
            if exp.project == query.get("project"):
                scores[exp.id] += 0.05

        ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [self._get(exp_id) for exp_id, _ in ranked[:top_k]]

    def _build_query_text(self, query: dict) -> str:
        return f"{query.get('error_type', '')} {query.get('error_message', '')} {query.get('task', '')}"
```

### 3.4 经验注入

```python
class ExperienceInjector:
    """将检索到的经验注入 Agent 上下文"""

    INJECTION_TEMPLATE = """## Relevant Past Experiences

The following similar issues were solved in previous tasks. Learn from these patterns:

{experiences}

## Instructions
- If a past experience matches your current bug EXACTLY, apply the same fix pattern.
- If similar but not identical, use it as guidance for your approach.
- ALWAYS verify the fix applies to your current code before editing.
"""

    def inject(self, experiences: list[Experience],
               context: list[dict]) -> list[dict]:
        if not experiences:
            return context

        formatted = []
        for i, exp in enumerate(experiences[:3], 1):
            formatted.append(f"""### Experience {i}: {exp.error_type}
- Bug: {exp.bug_signature[:200]}
- Fix: {exp.fix_description[:300]}
- Patch preview: ```diff\n{exp.fix_patch[:500]}\n```
- Success: {'yes' if exp.success else 'no'} ({exp.iterations_needed} iterations)
""")

        injection = self.INJECTION_TEMPLATE.format(
            experiences="\n".join(formatted)
        )

        # Insert as additional context message
        context.append({"role": "user", "content": injection})
        return context
```

### 3.5 经验记录触发

在 `DevAgentCore` 任务完成时自动记录：

```python
async def _record_experience(self, state: AgentLoopState):
    """任务完成后，提取经验并存入经验库"""
    if not self.agentic_config.get("enable_experience_library", True):
        return

    # 仅记录成功的修复任务
    if state.status != "COMPLETED" or not state.modified_files:
        return

    exp = Experience(
        id=f"exp_{state.task_id}",
        type="bug_fix",
        bug_signature=self._extract_bug_signature(state),
        error_type=self._extract_error_type(state),
        language=state.language,
        file_pattern=state.modified_files[0] if state.modified_files else "",
        code_context_before=self._extract_context(state),
        fix_description=self._extract_fix_description(state),
        fix_patch=self._extract_patch(state),
        success=True,
        iterations_needed=state.current_iteration,
        task_id=state.task_id,
        timestamp=datetime.now().isoformat(),
        project=os.path.basename(state.workspace),
        tags=self._extract_tags(state),
    )
    self.experience_library.add(exp)
```
