## 1. 概述

### 1.1 问题陈述

DevAgent V1 的测试能力存在以下不足：
- TestAgent 仅基于签名生成测试，不执行代码，无法发现运行时错误
- 测试生成是一次性的，不根据覆盖率反馈迭代
- 修复后仅做简单的回归测试，不验证边界条件
- 没有代码质量检查（linting、type checking）作为质量门

### 1.2 设计目标

- 实现覆盖率驱动的测试生成
- 修复后自动执行最小回归测试集
- 加入 lint/type-check 作为 Agentic Loop 中的即时反馈信号
- 引入 Mutation Testing 验证测试质量

## 2. 执行验证闭环

### 2.1 核心设计

```
         ┌─────────────┐
         │  修改代码    │
         └──────┬──────┘
                ▼
    ┌──────────────────────┐     失败
    │  即时验证（语法+Lint）│────────────▶ 修复
    └──────────┬───────────┘
               │ 通过
               ▼
    ┌──────────────────────┐     失败
    │  单元测试（快速）     │────────────▶ 故障定位
    └──────────┬───────────┘
               │ 通过
               ▼
    ┌──────────────────────┐     失败
    │  回归测试（完整）     │────────────▶ 故障定位
    └──────────┬───────────┘
               │ 通过
               ▼
    ┌──────────────────────┐     不足
    │  覆盖率检查           │────────────▶ 补充测试
    └──────────┬───────────┘
               │ 达标
               ▼
           完成
```

### 2.2 即时验证

```python
class InstantValidator:
    """每次代码修改后即时执行的最小验证集"""

    async def validate(self, modified_file: str) -> ValidationResult:
        results = []

        # 1. Python 语法检查（极快，~0.05s）
        syntax_ok = await self._check_syntax(modified_file)
        results.append(("syntax", syntax_ok))
        if not syntax_ok:
            return ValidationResult(False, results, "Syntax error")

        # 2. 导入检查（快，~0.2s）
        import_ok = await self._check_imports(modified_file)
        results.append(("import", import_ok))
        if not import_ok:
            return ValidationResult(False, results, "Import error")

        # 3. Ruff linting（快，~0.3s）
        lint_result = await self._run_lint(modified_file)
        results.append(("lint", lint_result))

        # 4. mypy type check（中速，~2s）
        type_result = await self._run_type_check(modified_file)
        results.append(("type_check", type_result))

        return ValidationResult(
            success=all(r[1].ok for r in results),
            checks=results
        )

    async def _check_syntax(self, file_path: str) -> CheckResult:
        try:
            code = Path(file_path).read_text()
            ast.parse(code)
            return CheckResult(ok=True)
        except SyntaxError as e:
            return CheckResult(ok=False, error=f"Line {e.lineno}: {e.msg}")

    async def _check_imports(self, file_path: str) -> CheckResult:
        """尝试导入模块以检查导入是否可解析"""
        proc = await asyncio.create_subprocess_exec(
            sys.executable, "-c",
            f"import ast; ast.parse(open('{file_path}').read())",
            stdout=PIPE, stderr=PIPE
        )
        await proc.communicate()
        return CheckResult(ok=proc.returncode == 0)

    async def _run_lint(self, file_path: str) -> CheckResult:
        """运行 Ruff 检查"""
        proc = await asyncio.create_subprocess_exec(
            "ruff", "check", file_path, "--output-format", "concise",
            stdout=PIPE, stderr=PIPE
        )
        stdout, _ = await asyncio.wait_for(proc.communicate(), timeout=10)
        issues = len(stdout.decode().splitlines()) if stdout else 0
        return CheckResult(ok=issues == 0, detail=f"{issues} lint issues")
```

### 2.3 覆盖率驱动的测试生成

```python
class CoverageDrivenTestGenerator:
    """基于覆盖率的迭代测试生成"""

    def generate_tests(self, source_files: list[str], existing_tests: list[str]) -> dict:
        """分析覆盖率缺口，生成补充测试"""
        # Step 1: 运行现有测试，收集覆盖率
        coverage_data = self._run_coverage(source_files, existing_tests)

        # Step 2: 识别未覆盖分支
        uncovered = self._identify_uncovered_branches(coverage_data)

        # Step 3: 对每个未覆盖分支生成测试
        new_tests = {}
        for branch in uncovered:
            test_code = self._generate_test_for_branch(branch)
            if test_code:
                new_tests[f"test_{branch.function}_{branch.branch_id}.py"] = test_code

        # Step 4: 验证新测试能通过
        verified_tests = self._verify_tests(new_tests)

        return verified_tests

    def _run_coverage(self, source_files, test_files) -> CoverageData:
        """使用 coverage.py 收集覆盖率数据"""
        with tempfile.NamedTemporaryFile(suffix=".py", delete=False) as tmp:
            tmp.write(b"")
            tmp_path = tmp.name

        subprocess.run([
            "python", "-m", "coverage", "run", "--source", ",".join(source_files),
            "-m", "pytest", *test_files, "-q"
        ], capture_output=True)

        # 解析 coverage.json
        result = subprocess.run(
            ["python", "-m", "coverage", "json", "-o", tmp_path],
            capture_output=True
        )
        with open(tmp_path) as f:
            data = json.load(f)
        os.unlink(tmp_path)

        return CoverageData.from_json(data)

    def _identify_uncovered_branches(self, coverage_data: CoverageData) -> list[Branch]:
        """识别未覆盖的分支"""
        uncovered = []
        for file_path, file_coverage in coverage_data.files.items():
            # 解析文件 AST，找到每个分支位置
            tree = ast.parse(Path(file_path).read_text())
            for node in ast.walk(tree):
                if isinstance(node, (ast.If, ast.For, ast.While, ast.Try)):
                    # 检查该分支是否被覆盖
                    branch_arcs = self._get_branch_arcs(node, file_coverage)
                    for arc in branch_arcs:
                        if not arc.executed:
                            uncovered.append(Branch(
                                file=file_path,
                                function=self._enclosing_function(node),
                                line=node.lineno,
                                branch_id=arc.id,
                                condition=self._get_condition_text(node)
                            ))
        return uncovered

    def _generate_test_for_branch(self, branch: Branch) -> str:
        """为特定未覆盖分支生成测试"""
        prompt = f"""Generate a pytest test case that covers the following uncovered branch:

File: {branch.file}
Function: {branch.function}
Line: {branch.line}
Branch condition: {branch.condition}

Context (surrounding code):
```python
{branch.surrounding_code}
```

Generate a test that exercises the UNcovered branch. Return ONLY the test function code."""

        test_code = self.llm.chat(
            messages=[{"role": "user", "content": prompt}]
        )
        return self._clean_test_code(test_code)
```

### 2.4 智能回归测试选择

```python
class SmartRegressionSelector:
    """根据代码变更，仅选择受影响的测试子集"""

    def select(self, modified_files: list[str], test_files: list[str]) -> list[str]:
        """选择与修改相关的测试"""
        # Method 1: AST 依赖分析
        affected_tests = set()

        for test_file in test_files:
            test_ast = ast.parse(Path(test_file).read_text())
            imports_from_modified = self._find_imports_from(test_ast, modified_files)
            if imports_from_modified:
                affected_tests.add(test_file)

        # Method 2: 基于 git diff 的测试选择
        diff_files = self._get_diff_files()
        for modified in modified_files:
            # 查找测试文件中 import modified 的
            for test_file in test_files:
                if self._test_imports_module(test_file, modified):
                    affected_tests.add(test_file)

        return sorted(affected_tests) if affected_tests else test_files  # fallback to all

    def _find_imports_from(self, tree: ast.AST, modified_files: list[str]) -> bool:
        """检查 AST 是否 import 了被修改的模块"""
        modified_modules = {
            str(Path(f).with_suffix('')).replace('/', '.')
            for f in modified_files
        }
        for node in ast.walk(tree):
            if isinstance(node, ast.ImportFrom):
                if node.module and any(
                    node.module.startswith(m) or m.startswith(node.module)
                    for m in modified_modules
                ):
                    return True
        return False
```

### 2.5 变异测试（Mutation Testing）

```python
class MutationTester:
    """通过注入 bug 验证测试质量"""

    MUTATION_OPERATORS = {
        "Arithmetic": ["+", "-", "*", "/", "%"],
        "Comparison": ["==", "!=", "<", ">", "<=", ">="],
        "Logical": ["and", "or", "not"],
        "Boundary": [" < ", " <= "],  # 边界条件变异
    }

    def test_quality(self, source_file: str, test_file: str) -> MutationReport:
        """运行变异测试，报告测试套件的查杀能力"""
        original_code = Path(source_file).read_text()
        mutants = self._generate_mutants(original_code)
        killed = 0
        survived = []

        for mutant in mutants:
            # 应用变异
            Path(source_file).write_text(mutant.code)

            # 运行测试
            result = subprocess.run(
                ["python", "-m", "pytest", test_file, "-q"],
                capture_output=True, timeout=30
            )

            if result.returncode != 0:
                killed += 1  # 测试成功捕获变异
            else:
                survived.append(mutant)

            # 恢复原始代码
            Path(source_file).write_text(original_code)

        mutation_score = killed / len(mutants) if mutants else 1.0

        return MutationReport(
            total_mutants=len(mutants),
            killed=killed,
            survived=survived,
            mutation_score=mutation_score,
            quality="good" if mutation_score > 0.8 else "needs_improvement"
        )
```

### 3.1 多级质量门

class QualityGateSystem:
    """重构后的多级质量门"""

    GATES = {
        "L1_SYNTAX": {
            "name": "Syntax Valid",
            "check": lambda s: s.syntax_errors == 0,
            "blocking": True,   # 不通过则阻止继续
            "auto_fix": False    # 不能自动修复
        },
        "L2_LINT": {
            "name": "Lint Clean",
            "check": lambda s: s.lint_issues <= 5,
            "blocking": False,
            "auto_fix": True     # 可以自动修复
        },
        "L3_TYPE": {
            "name": "Type Check",
            "check": lambda s: s.type_errors == 0,
            "blocking": False,
            "auto_fix": False
        },
        "L4_UNIT_TESTS": {
            "name": "Unit Tests Pass",
            "check": lambda s: s.unit_failed == 0,
            "blocking": True,
            "auto_fix": True     # 进入修复循环
        },
        "L5_COVERAGE": {
            "name": "Coverage >= 80%",
            "check": lambda s: s.coverage >= 0.8,
            "blocking": False,
            "auto_fix": True     # 自动生成补充测试
        },
        "L6_REGRESSION": {
            "name": "Regression Tests Pass",
            "check": lambda s: s.regression_failed == 0,
            "blocking": True,
            "auto_fix": True
        }
    }
